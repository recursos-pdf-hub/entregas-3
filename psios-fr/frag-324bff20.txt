    window.addEventListener("resize", function () { measure(); build(); });
    window.addEventListener("orientationchange", function () { measure(); build(); });
    window.addEventListener("load", function () { measure(); build(); });
    track.querySelectorAll("img").forEach(function (img) {
      if (!img.complete) img.addEventListener("load", function () { measure(); build(); });
    });

    // 4. loop em pixels
    function frame(now) {
      if (last === null) last = now;
      var dt = (now - last) / 1000;
      last = now;
      if (dt > 0.25) dt = 0;
      if (!paused && groupWidth > 0) {
        x -= speed * dt;
        while (x <= -groupWidth) x += groupWidth;
        track.style.transform = "translate3d(" + x + "px,0,0)";
      }
      requestAnimationFrame(frame);
    }
    if (!reduceMotion) requestAnimationFrame(frame);

    marquee.addEventListener("click", function () { paused = !paused; });
    marquee.addEventListener("touchstart", function () { paused = true; }, { passive: true });
    marquee.addEventListener("touchend", function () { paused = false; }, { passive: true });
    marquee.addEventListener("mouseenter", function () { paused = true; });
    marquee.addEventListener("mouseleave", function () { paused = false; });
    document.addEventListener("visibilitychange", function () { last = null; });
  }

  
  var revealables = document.querySelectorAll(".ps-reveal");
  if (reduceMotion || !("IntersectionObserver" in window)) {
    revealables.forEach(function (el) { el.classList.add("ps-is-visible"); });
  } else {
    var io = new IntersectionObserver(function (entries) {
      entries.forEach(function (entry) {
        if (entry.isIntersecting) {
          entry.target.classList.add("ps-is-visible");
          io.unobserve(entry.target);
        }
      });
    }, { rootMargin: "0px 0px -8% 0px", threshold: 0.08 });
    revealables.forEach(function (el) { io.observe(el); });
  }
})();
